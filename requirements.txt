# --- Optional: RTMPose3D backend (the "RTMPose3D" dropdown option) ----------

#   pip install rtmlib onnxruntime          # CPU / Apple Silicon (not real-time)

# For onnxruntime-gpu the CUDA 12 + cuDNN 9 runtime DLLs must be importable;
# installing PyTorch (which bundles them) satisfies this and the backend adds
# torch/lib to the DLL search path automatically. See HANDOFF.md.
